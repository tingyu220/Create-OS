"""冻结证据上的只读专家循环；不持有 Runtime、文件路径或写入能力。"""

from copy import deepcopy
from dataclasses import asdict
import json
from uuid import uuid4

from creative_os.combat_model import CombatRequest, CombatResult
from creative_os.llm_writer import ModelMessage


MAX_STEPS = 4
MAX_READS = 8
MAX_SOURCE_CHARS = 12000
MAX_TOTAL_CHARS = 48000
MAX_INPUT_CHARS = 100000
PROTOCOL = """你是战斗编排专家。只研究本次冻结来源，提出自由文本的叙事建议，不把未来计划、未确认候选或邻章当作目标章事实。
按需选择读取顺序，可判断非战斗冲突更合适，或材料不足。目标正文缺失必须说明未知，不能推断章内缺陷。
只能输出一个 JSON 工具动作：
{"action":"read_source","source_id":"目录内引用"}
{"action":"finish","verdict":"combat/setup/noncombat/insufficient四选一","reply":"自然语言建议","alternatives":[],"foreshadowing":[],"evidence_refs":[],"unknowns":[]}
evidence_refs 只能引用实际读到正文的 source_id。所有数组元素是字符串；身份由服务器注入。
无权确认、发布、改稿、访问任意路径或再次委派；最多4步，尽早完成。"""


def source_catalog(request):
    """目录仅暴露冻结元数据，正文必须经 read_source 读取。"""
    return [{key: deepcopy(value) for key, value in source.items() if key != "text"} for source in request.sources]


def _decode(raw):
    try:
        action = json.loads(raw)
    except (TypeError, ValueError) as error:
        raise ValueError("combat_model_json_invalid") from error
    fields = {"read_source": {"action", "source_id"}, "finish": {"action", "verdict", "reply", "alternatives", "foreshadowing", "evidence_refs", "unknowns"}}
    if (not isinstance(action, dict) or not isinstance(action.get("action"), str)
            or action["action"] not in fields or set(action) != fields[action["action"]]):
        raise ValueError("combat_tool_not_allowed")
    return action


def run_combat_expert(request, *, execute_step, read_source) -> CombatResult:
    if not isinstance(request, CombatRequest):
        raise ValueError("combat_request_invalid")
    # 回调得到的材料也与原请求隔离，不能就地更改嵌套冻结字段。
    request = deepcopy(request)
    header = {"instruction": request.instruction, "scope": asdict(request.scope), "sources": source_catalog(request)}
    messages = [ModelMessage("system", PROTOCOL), ModelMessage("user", json.dumps(header, ensure_ascii=False))]
    read_ids, reads, total = [], 0, 0

    def result(verdict, reply, alternatives=(), foreshadowing=(), evidence_refs=(), unknowns=()):
        return CombatResult(f"combat-result-{uuid4().hex}", request.project_id, request.job_id, verdict, reply,
                            tuple(alternatives), tuple(foreshadowing), tuple(evidence_refs), tuple(unknowns), request.scope)

    def exhausted(reason):
        return result("insufficient", "本轮专家预算已用完；未能完成的判断仍未知，请缩小范围。",
                      evidence_refs=read_ids, unknowns=(reason,))

    for step in range(1, MAX_STEPS + 1):
        if sum(len(item.content) for item in messages) > MAX_INPUT_CHARS:
            return exhausted("单步输入预算耗尽")
        raw = execute_step(tuple(messages), step=step, sources=tuple(read_ids))
        action = _decode(raw)
        if action["action"] == "finish":
            for key in ("alternatives", "foreshadowing", "evidence_refs", "unknowns"):
                if not isinstance(action[key], list) or any(not isinstance(item, str) for item in action[key]):
                    raise ValueError("combat_result_invalid")
            if not isinstance(action["reply"], str) or not action["reply"].strip():
                raise ValueError("combat_reply_invalid")
            if not set(action["evidence_refs"]) <= set(read_ids):
                raise ValueError("combat_evidence_not_read")
            return result(**{key: value for key, value in action.items() if key != "action"})
        source_id = action["source_id"]
        if not isinstance(source_id, str) or not any(item["source_id"] == source_id for item in request.sources):
            raise ValueError("combat_source_not_found")
        if reads >= MAX_READS:
            return exhausted("来源读取次数预算耗尽")
        source = read_source(deepcopy(request), source_id)
        reads += 1
        # 读取回调也不能扩大冻结权限或偷换正文。
        frozen = next(item for item in request.sources if item["source_id"] == source_id)
        if source != frozen:
            raise ValueError("combat_source_snapshot_mismatch")
        text = source.get("text", "")
        if len(text) > MAX_SOURCE_CHARS or total + len(text) > MAX_TOTAL_CHARS:
            return exhausted("来源字符预算耗尽")
        total += len(text)
        if source.get("usage") == "included" and source_id not in read_ids:
            read_ids.append(source_id)
        messages.extend([ModelMessage("assistant", raw), ModelMessage("user", "只读工具结果：" + json.dumps(source, ensure_ascii=False))])
    return exhausted("专家步骤预算耗尽")

"""主 Agent 的有限讨论工具及只读专家桥接；不创建客户端或队列。"""

from copy import deepcopy
from dataclasses import asdict
import json

from creative_os.combat_agent import run_combat_expert
from creative_os.combat_context import build_combat_context, read_combat_source
from creative_os.combat_model import CombatRequest, CombatScope
from creative_os.llm_writer import ModelMessage
from creative_os.project_directory import validated_project
from creative_os.writer_context_sources import ContextCapture


MAX_STEPS = 6
MAX_INPUT_CHARS = 100000
PROTOCOL = """你可自主决定是否需要战斗编排专家，普通分析可直接结束，不要求每章委派。
每次至多输出一个工具 JSON：{"action":"consult_combat"} 或 {"action":"finish","reply":"给作者的自然语言分析"}。
每轮最多委派一次，不能递归。专家结果仅是建议，无权确认、发布或修改正文。保留普通自然语言分析输出的兼容性。"""


def freeze_combat_request(root, project_id, job, *, original=None):
    if original is not None:
        return deepcopy(original.combat_context)
    # 旧无 metadata 的临时/传统入口保持普通分析兼容，不能获得专家读权限。
    if not (root / "project.json").exists():
        if job.requested_expert:
            raise ValueError("combat_project_identity_required")
        return None
    selected, _ = validated_project(root.parent, project_id)
    if selected != root.resolve():
        raise ValueError("combat_project_scope_mismatch")
    scope = CombatScope(() if job.chapter_number is None else (job.chapter_number,), "本轮故事研究" if job.chapter_number is None else "目标章节冲突研究")
    try:
        frozen = build_combat_context(selected, project_id, scope, task_id=job.job_id,
                                      session_id=job.session_id, instruction=job.instruction)
    except ValueError as error:
        if not str(error).startswith("required_combat_context_budget_exceeded"):
            raise
        # 普通分析仍可进行；显式或自主委派消费同一冻结失败，不能临时换新材料。
        return {"error": str(error)}
    return json.loads(json.dumps(asdict(frozen), ensure_ascii=False))


def consult_combat(runtime, job, *, main_step):
    current = runtime.get(job.job_id)
    if current.combat_results:
        raise ValueError("combat_delegation_limit")
    raw = deepcopy(job.combat_context)
    if raw is None:
        raise ValueError("combat_context_unavailable")
    if "error" in raw:
        raise ValueError(raw["error"])
    raw["scope"] = CombatScope(tuple(raw["scope"]["chapter_numbers"]), raw["scope"]["purpose"])
    raw["sources"] = tuple(raw["sources"])
    request = CombatRequest(**raw)

    def execute(messages, *, step, sources):
        capture = ContextCapture()
        # 正文在完整消息中原样捕获；引用登记准确区分已读与仅目录。
        for source in request.sources:
            capture.add("战斗冻结来源", source["source_ref"], reason="read" if source["source_id"] in sources else "catalog_only",
                        evidence_ref=source["source_id"], hash=source.get("hash"), version=source.get("version"),
                        usage="recorded" if source["source_id"] in sources else "not_in_model")
        return runtime._execute_story_step(job, messages, capture, f"combat:{main_step}:{step}")

    result = run_combat_expert(request, execute_step=execute, read_source=read_combat_source)
    # 重试只复用原输入身份；结果属于这次任务，不能冒充原失败任务结果。
    value = {**asdict(result), "job_id": job.job_id}
    runtime._set(job.job_id, combat_results=(*current.combat_results, value))
    return value


def run_chapter_turn(runtime, job, messages, capture):
    frozen = job.main_input
    if frozen:
        messages = [ModelMessage(item["role"], item["content"]) for item in frozen]
        # 重试的段偏移也来自首次装配，不能与最新上下文混用。
        capture.entries = deepcopy((job.analysis_state or {}).get("capture_entries", []))
    else:
        messages = list(messages)
        messages[0] = ModelMessage("system", messages[0].content + "\n" + PROTOCOL)
        runtime._set(job.job_id, main_input=tuple(asdict(item) for item in messages))
    trace = []
    if job.requested_expert == "combat":
        value = consult_combat(runtime, job, main_step=0)
        trace.append({"step": 0, "input": {"action": "consult_combat"}, "result": value, "selection": "explicit"})
        messages.append(ModelMessage("user", "专家工具结果：" + json.dumps(value, ensure_ascii=False)))
        capture = ContextCapture()
        runtime._set(job.job_id, tool_trace=tuple(trace))
    for step in range(1, MAX_STEPS + 1):
        if sum(len(item.content) for item in messages) > MAX_INPUT_CHARS:
            return "本轮输入预算已用完，剩余问题未知，请缩小范围。"
        raw = runtime._execute_story_step(job, messages, capture, step)
        try:
            action = json.loads(raw)
        except (TypeError, ValueError):
            # 兼容已有直接分析文本；貌似工具 JSON 的错误不能降级成成功。
            if str(raw).lstrip().startswith(("{", "[")):
                raise ValueError("chapter_model_json_invalid")
            action = {"action": "finish", "reply": raw}
        if not isinstance(action, dict):
            raise ValueError("chapter_tool_not_allowed")
        if action.get("action") == "finish" and set(action) == {"action", "reply"}:
            if not isinstance(action["reply"], str) or not action["reply"].strip():
                raise ValueError("chapter_reply_invalid")
            trace.append({"step": step, "input": action, "result": "finished"})
            runtime._set(job.job_id, tool_trace=tuple(trace))
            return action["reply"]
        if action != {"action": "consult_combat"}:
            raise ValueError("chapter_tool_not_allowed")
        value = consult_combat(runtime, job, main_step=step)
        trace.append({"step": step, "input": action, "result": value})
        runtime._set(job.job_id, tool_trace=tuple(trace))
        messages.extend([ModelMessage("assistant", raw), ModelMessage("user", "专家工具结果：" + json.dumps(value, ensure_ascii=False))])
        capture = ContextCapture()
    return "本轮工具步骤预算已用完；未完成的判断仍未知，请缩小范围或继续讨论。"


def freeze_analysis_state(job, context, formal_word_count, capture):
    decision = {"decision_id": job.decision_id or context.decision.decision_id, "intent": job.instruction,
                "target_chapter": job.chapter_number, "context_refs": list(context.source_refs),
                "created_at": context.decision.created_at, "target_words": job.target_words,
                "tolerance_words": job.tolerance_words, "min_words": job.min_words, "max_words": job.max_words,
                "formal_word_count": formal_word_count, "project_id": job.project_id,
                "session_id": job.session_id, "task_id": job.job_id}
    return {"decision": decision, "capture_entries": deepcopy(capture.entries)}


def replay_chapter_analysis(runtime, job):
    """已送达模型的分析重试直接回放旧装配，不再触碰最新正文、草稿或经验。"""
    content = run_chapter_turn(runtime, job, (), ContextCapture())
    decision = {**deepcopy(job.analysis_state["decision"]), "task_id": job.job_id, "analysis": str(content)}
    runtime._write_decision(decision["decision_id"], decision)
    runtime._set(job.job_id, status="completed", message="分析已完成，正文未修改。",
                 decision_id=decision["decision_id"], reply=content)

{"action":"consult_combat"}
{"action":"finish","reply":"给作者的自然语言说明、问题或备选"}
{"action":"finish_turn","reply":"给作者的自然语言说明、问题或备选"}
可连续调用多个工具，每步一个 JSON；需要作者补充时 finish_turn。不得输出 confirm/publish 等其它动作。"""


def decode_action(text):
    try:
        action = json.loads(text)
    except (TypeError, ValueError) as error:
        raise ValueError("story_model_json_invalid") from error
    fields = {"read_candidate": {"action"}, "read_confirmed": {"action"},
              "propose_revision": {"action", "expected_revision", "blocks", "open_questions"},
              "finish_turn": {"action", "reply"}, "finish": {"action", "reply"}, "consult_combat": {"action"}}
    if not isinstance(action, dict) or not isinstance(action.get("action"), str) or action["action"] not in fields or set(action) != fields[action["action"]]:
        raise ValueError("story_tool_not_allowed")
    if action["action"] in {"finish_turn", "finish"} and (not isinstance(action["reply"], str) or not action["reply"].strip()):
        raise ValueError("story_reply_invalid")
    return action


def run_story_turn(runtime, job):
    runtime._set(job.job_id, status="thinking", message="正在结合 Idea、讨论历史和当前设定思考。")
    history = runtime.conversations.conversation(None)["messages"]
    # 历史含本轮原文；失败重试不会另造作者消息。
    if job.main_input:
        frozen = [ModelMessage(item["role"], item["content"]) for item in job.main_input]
        messages, state_message = frozen[:-1], frozen[-1]
        state = state_message.content.removeprefix("本步领域状态（候选不是正式事实）：\n")
    else:
        messages = [ModelMessage("system", PROTOCOL + "\n" + EXPERT_PROTOCOL)] + [ModelMessage(item["role"], item["content"]) for item in history]
        view = runtime.story.view()
        state = json.dumps({"project_id": runtime.project_id, "session_id": job.session_id, "candidate": view["candidate"], "confirmed": view["confirmed"]}, ensure_ascii=False)
        state_message = ModelMessage("user", "本步领域状态（候选不是正式事实）：\n" + state)
        runtime._set(job.job_id, main_input=tuple(asdict(item) for item in [*messages, state_message]))
    trace = []
    frozen_state = json.loads(state)
    if job.requested_expert == "combat":
        value = consult_combat(runtime, job, main_step=0)
        trace.append({"step": 0, "input": {"action": "consult_combat"}, "result": value, "selection": "explicit"})
        messages.append(ModelMessage("user", "专家工具结果：" + json.dumps(value, ensure_ascii=False)))
        runtime._set(job.job_id, tool_trace=tuple(trace))
    for step in range(1, MAX_STEPS + 1):
        step_messages = [*messages, state_message]
        if sum(len(message.content) for message in step_messages) > MAX_INPUT_CHARS:
            raise ValueError("story_input_budget_exceeded_continue_with_shorter_context")
        capture = ContextCapture()
        capture.add("本步故事候选与正式设定", f"story-state:{runtime.project_id}:{step}", state, version=step, message_start=len("本步领域状态（候选不是正式事实）：\n"))
        raw = runtime._execute_story_step(job, step_messages, capture, step)
        action = decode_action(raw)
        name = action["action"]
        if name in {"finish_turn", "finish"}:
            trace.append({"step": step, "input": action, "result": "finished"})
            runtime._set(job.job_id, status="completed", message="本轮讨论已完成，候选可编辑；正式设定须作者明确确认。", tool_trace=tuple(trace), reply=action["reply"])
            return
        if name == "consult_combat":
            result = consult_combat(runtime, job, main_step=step)
        elif name == "read_candidate":
            result = frozen_state["candidate"]
        elif name == "read_confirmed":
            result = frozen_state["confirmed"]
        else:
            result = runtime.story.propose(job.session_id, action["blocks"], action["open_questions"],
                                           expected_revision=action["expected_revision"], actor="agent",
                                           operation_id=f"agent:{job.job_id}:step:{step}")
            frozen_state["candidate"] = result
            state = json.dumps(frozen_state, ensure_ascii=False)
            state_message = ModelMessage("user", "本步领域状态（候选不是正式事实）：\n" + state)
        trace.append({"step": step, "input": action, "result": result})
        runtime._set(job.job_id, tool_trace=tuple(trace), message=f"已执行 {name}，继续判断下一步。")
        messages.extend([ModelMessage("assistant", raw), ModelMessage("user", "工具结果：" + json.dumps(result, ensure_ascii=False))])
    runtime._set(job.job_id, status="completed", message="本轮工具步数已用完，可继续讨论；已有候选已保留。", tool_trace=tuple(trace),
                 reply="本轮已达到工具步数预算。已保留讨论记录和有效候选，请补充想法或发送“继续讨论”。")

class DelegationClient:
    def __init__(self, *, autonomous=False, error=None):
        self.autonomous, self.error, self.calls = autonomous, error, []

    def complete(self, messages, **kwargs):
        self.calls.append((deepcopy(messages), kwargs))
        expert = "你是战斗编排专家" in messages[0].content
        if expert:
            if self.error:
                raise self.error
            return json.dumps(conclusion("insufficient"), ensure_ascii=False)
        consulted = any("combat-result-" in item.content for item in messages)
        return json.dumps({"action": "consult_combat"} if self.autonomous and not consulted
                          else {"action": "finish", "reply": "讨论结果"}, ensure_ascii=False)

CONTROLLED_CALLER = "WriterAgentRuntime._execute_story_step"


def validate_combat_loops(sources):
    """领域循环仅调用注入单步；保留原透明适配器的更严格检查。"""
    validate_transparent_delegate(sources)
    for module, name, maximum in (("combat_agent", "run_combat_expert", 4),
                                  ("chapter_agent_tools", "run_chapter_turn", 6),
                                  ("story_architecture_agent", "run_story_turn", 6)):
        tree = ast.parse(sources[f"creative_os/{module}.py"])
        function = qualified_definitions(tree)[name]
        constants = {node.targets[0].id: node.value.value for node in tree.body
                     if isinstance(node, ast.Assign) and len(node.targets) == 1
                     and isinstance(node.targets[0], ast.Name) and isinstance(node.value, ast.Constant)}
        assert constants.get("MAX_STEPS") == maximum, "unexpected domain step budget"
        assert not any(isinstance(node, (ast.While, ast.AsyncFor)) for node in ast.walk(tree)), "unbounded domain loop"
        loops = [node for node in ast.walk(function) if isinstance(node, ast.For)
                 and ast.unparse(node.iter) == "range(1, MAX_STEPS + 1)"]
        assert len(loops) == 1, "missing bounded domain loop"
        for call in (node for node in ast.walk(tree) if isinstance(node, ast.Call)):
            called = ast.unparse(call.func)
            assert not called.endswith((".complete", ".execute", ".client_factory", ".from_env")), "domain creates a model exit"
            assert "RuntimeRunner" not in called and "CapturingModelAdapter" not in called, "domain owns runtime adapter"


def qualified_definitions(tree):
    """使用类和函数的词法路径，避免不同类的同名方法互相覆盖。"""
    definitions = {}

    def visit(node, prefix=()):
        if isinstance(node, (ast.ClassDef, ast.FunctionDef, ast.AsyncFunctionDef)):
            prefix = (*prefix, node.name)
            key = ".".join(prefix)
            assert key not in definitions, f"ambiguous AST definition: {key}"
            definitions[key] = node
        for child in ast.iter_child_nodes(node):
            visit(child, prefix)

    visit(tree)
    return definitions


def signature(name, node):
    args = node.args
    parts = [arg.arg for arg in args.posonlyargs]
    if parts:
        parts.append("/")
    parts.extend(arg.arg for arg in args.args)
    if args.vararg:
        parts.append("*" + args.vararg.arg)
    elif args.kwonlyargs:
        parts.append("*")
    parts.extend(arg.arg for arg in args.kwonlyargs)
    if args.kwarg:
        parts.append("**" + args.kwarg.arg)
    return f"{name}({','.join(parts)})"


def module_name(path):
    parts = list(PurePosixPath(path).with_suffix("").parts)
    return ".".join(parts[:-1] if parts[-1] == "__init__" else parts)


def _imports(tree, module, is_package):
    result = {}

    def bind(name, symbol):
        previous = result.get(name)
        related = lambda value: value is not None and (value.endswith((".CapturingModelAdapter", ".RuntimeRunner")) or ADAPTER.startswith(value + ".") or RUNNER.startswith(value + "."))
        assert previous in {None, symbol} or not (related(previous) or related(symbol)), "ambiguous protected import alias"
        result[name] = symbol

    for node in ast.walk(tree):
        if isinstance(node, ast.Import):
            for alias in node.names:
                bind(alias.asname or alias.name.split(".")[0], alias.name if alias.asname else alias.name.split(".")[0])
        elif isinstance(node, ast.ImportFrom):
            base = node.module or ""
            if node.level:
                package = module.split(".") if is_package else module.split(".")[:-1]
                base = ".".join(package[:len(package) - node.level + 1] + ([base] if base else []))
            for alias in node.names:
                assert not (alias.name == "*" and base in {"creative_os", "creative_os.writer_context_sources", "creative_os.agent_runtime"}), "delegate wildcard import"
                bind(alias.asname or alias.name, f"{base}.{alias.name}")
    return result


def _without_docstring(node):
    node = deepcopy(node)
    for item in ast.walk(node):
        if isinstance(item, (ast.ClassDef, ast.FunctionDef)) and item.body and isinstance(item.body[0], ast.Expr) and isinstance(item.body[0].value, ast.Constant) and isinstance(item.body[0].value.value, str):
            item.body.pop(0)
    return node


def validate_transparent_delegate(sources):
    trees = {path: ast.parse(source) for path, source in sources.items()}
    actual_class = qualified_definitions(trees[ADAPTER_FILE])["CapturingModelAdapter"]
    expected_class = ast.parse("""class CapturingModelAdapter:
    def __init__(self, client, capture):
        self.client, self.capture = client, capture
    def complete(self, messages, **kwargs):
        self.capture(messages)
        return self.client.complete(messages, **kwargs)
""").body[0]
    # 整个透明类限定为原样保存依赖、先观察再一次委派；拒绝额外路径及改参。
    assert ast.dump(_without_docstring(actual_class)) == ast.dump(expected_class), "delegate is not transparent"
    imports = {path: _imports(tree, module_name(path), PurePosixPath(path).stem == "__init__") for path, tree in trees.items()}
    exports = {f"{module_name(path)}.{name}": symbol for path, aliases in imports.items() for name, symbol in aliases.items()}

    def canonical(symbol):
        # 解析导入别名和已知模块再导出，不推断任意动态赋值。
        seen = set()
        while symbol not in seen:
            seen.add(symbol)
            parts = symbol.split(".")
            for index in range(len(parts), 0, -1):
                key = ".".join(parts[:index])
                if key in exports:
                    replacement = ".".join([exports[key], *parts[index:]])
                    if replacement != symbol:
                        symbol = replacement
                        break
            else:
                return symbol
        raise AssertionError("delegate import cycle")

    delegate_modules = {ADAPTER.rsplit(".", 1)[0]}
    delegate_modules.update(name.rsplit(".", 1)[0] for name, value in exports.items() if canonical(value) == ADAPTER)

    def delegate_namespace(symbol):
        return any(module == symbol or module.startswith(symbol + ".") for module in delegate_modules)

    constructions = {CALLER: 0, CONTROLLED_CALLER: 0}
    for path, tree in trees.items():
        aliases = imports[path]
        parents = {child: parent for parent in ast.walk(tree) for child in ast.iter_child_nodes(parent)}
        definitions = qualified_definitions(tree)
        function_names = {id(node): name for name, node in definitions.items() if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef))}

        def resolve(node):
            if isinstance(node, ast.Name):
                return canonical(aliases.get(node.id, node.id))
            if isinstance(node, ast.Attribute):
                return canonical(f"{resolve(node.value)}.{node.attr}")
            return ""

        def protected(symbol):
            return symbol in {ADAPTER, RUNNER} or delegate_namespace(symbol) or RUNNER.startswith(symbol + ".")

        # 类的直接导入或跨模块再导出，仅允许既有 Writer 模块消费。
        for symbol in aliases.values():
            if canonical(symbol) == ADAPTER:
                assert path == CALLER_FILE, "delegate imported outside Writer"
        for node in ast.walk(tree):
            if isinstance(node, ast.arg) and node.arg in aliases:
                assert not protected(canonical(aliases[node.arg])), "delegate/Runner import shadowed by argument"
            if isinstance(node, ast.ExceptHandler) and node.name in aliases:
                assert not protected(canonical(aliases[node.name])), "delegate/Runner import shadowed by exception"
            if isinstance(node, (ast.ClassDef, ast.FunctionDef, ast.AsyncFunctionDef)) and node.name in aliases:
                assert not protected(canonical(aliases[node.name])), "delegate/Runner import shadowed by definition"
            if not isinstance(node, (ast.Name, ast.Attribute)):
                continue
            symbol = resolve(node)
            if isinstance(node.ctx, (ast.Store, ast.Del)):
                assert not protected(symbol), "delegate/Runner import reassigned"
            if symbol.startswith(ADAPTER + "."):
                raise AssertionError("delegate method reference escaped")
            if symbol == ADAPTER:
                constructor = parents.get(node)
                assert isinstance(constructor, ast.Call) and constructor.func is node, "delegate class escaped"
                assert path == CALLER_FILE, "delegate constructed outside Writer"
                owner, cursor = None, constructor
                while cursor in parents:
                    cursor = parents[cursor]
                    assert not isinstance(cursor, (ast.For, ast.AsyncFor, ast.While, ast.comprehension)), "delegate construction in loop"
                    if isinstance(cursor, (ast.FunctionDef, ast.AsyncFunctionDef)):
                        owner = function_names[id(cursor)]
                        break
                assert owner in constructions, "delegate constructed outside declared Writer entry"
                outer = parents.get(constructor)
                # 仅允许直接占据 execute 的 adapter 参数，禁止赋值、转交或直接 complete。
                if isinstance(outer, ast.keyword):
                    assert outer.arg == "adapter", "delegate in wrong keyword"
                    outer = parents.get(outer)
                    adapter_argument = True
                else:
                    adapter_argument = isinstance(outer, ast.Call) and len(outer.args) >= 2 and outer.args[1] is constructor
                assert adapter_argument and isinstance(outer, ast.Call), "delegate not immediate adapter argument"
                method = outer.func
                assert isinstance(method, ast.Attribute) and method.attr == "execute", "delegate sent to other entry"
                assert isinstance(method.value, ast.Call) and resolve(method.value.func) == RUNNER, "delegate not passed to RuntimeRunner"
                constructions[owner] += 1
            elif symbol and delegate_namespace(symbol):
                parent = parents.get(node)
                assert isinstance(parent, ast.Attribute) and parent.value is node, "delegate module escaped"
    assert all(count == 1 for count in constructions.values()), "Writer must have exactly one delegate construction per declared entry"
